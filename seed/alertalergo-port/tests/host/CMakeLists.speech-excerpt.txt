
# FSD §18.2 FR-SPK-04 — o eSpeak NG vendorizado, compilado no host, contra a
# referência do upstream (scripts/gen-espeak.sh). Mesmas fontes do firmware.
set(ESPK ${COMPONENTS}/third_party/espeak-ng)
set(ESPK_U ${ESPK}/upstream/src/libespeak-ng)
set(ESPK_D ${ESPK}/upstream/src/ucd-tools/src)
add_library(espeak_host STATIC
    ${ESPK_U}/common.c ${ESPK_U}/mnemonics.c ${ESPK_U}/error.c ${ESPK_U}/ieee80.c
    ${ESPK_U}/compiledata.c ${ESPK_U}/compiledict.c
    ${ESPK_U}/dictionary.c ${ESPK_U}/encoding.c ${ESPK_U}/intonation.c ${ESPK_U}/langopts.c
    ${ESPK_U}/numbers.c ${ESPK_U}/phoneme.c ${ESPK_U}/phonemelist.c ${ESPK_U}/readclause.c
    ${ESPK_U}/setlengths.c ${ESPK_U}/soundicon.c ${ESPK_U}/spect.c ${ESPK_U}/ssml.c
    ${ESPK_U}/synthdata.c ${ESPK_U}/synthesize.c ${ESPK_U}/tr_languages.c ${ESPK_U}/translate.c
    ${ESPK_U}/translateword.c ${ESPK_U}/voices.c ${ESPK_U}/wavegen.c ${ESPK_U}/speech.c ${ESPK_U}/espeak_api.c
    ${ESPK_D}/case.c ${ESPK_D}/categories.c ${ESPK_D}/ctype.c ${ESPK_D}/proplist.c ${ESPK_D}/scripts.c ${ESPK_D}/tostring.c)
target_include_directories(espeak_host PUBLIC ${ESPK}/upstream/src/include
    PRIVATE ${ESPK}/config ${ESPK_U} ${ESPK_D}/include)
target_compile_options(espeak_host PRIVATE -w -Wno-error -fwrapv)
target_compile_definitions(espeak_host PUBLIC LIBESPEAK_NG_EXPORT=1 PRIVATE PATH_ESPEAK_DATA="/es")
target_link_libraries(espeak_host PUBLIC m)
add_executable(speech_pcm speech_pcm.c)
target_compile_options(speech_pcm PRIVATE -Wno-error)
target_include_directories(speech_pcm PRIVATE ${COMPONENTS}/l1_speech/include)
target_link_libraries(speech_pcm espeak_host)
add_test(NAME TC-SPK-04 COMMAND python3 ${CMAKE_CURRENT_LIST_DIR}/check_speech_pcm.py
         $<TARGET_FILE:speech_pcm> ${ESPK}/data ${CMAKE_CURRENT_BINARY_DIR}/speech_pcm_out)
